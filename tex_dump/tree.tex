\begin{split}
f = x + y
\end{split}

\textbf{Оптимизация формулы:}
\begin{align*}
f = x + y\end{align*}

\textbf{Дифференцирование формулы:}
\begin{align*}
 \frac {\partial f} {\partial x}  = 1 + 0\end{align*}

\textbf{И снова оптимизация формулы:}
\begin{align*}
 \frac {\partial f} {\partial x}  = 1\end{align*}

\textbf{Дифференцирование формулы:}
\begin{align*}
 \frac {\partial^2 f} {\partial x \cdot \partial y}  = 0\end{align*}

\textbf{И снова оптимизация формулы:}
\begin{align*}
 \frac {\partial^2 f} {\partial x \cdot \partial y}  = 0\end{align*}
