f = \sin(z) + x \cdot y

\textbf{Оптимизация формулы:}
\begin{align*}
f = \sin(z) + x \cdot y\end{align*}

\textbf{Дифференцирование формулы:}
\begin{align*}
 \frac {\partial f} {\partial x}  = \cos(z) \cdot  \frac {\partial z} {\partial x}  + 1 \cdot y + x \cdot 0\end{align*}

\textbf{И снова оптимизация формулы:}
\begin{align*}
 \frac {\partial f} {\partial x}  = \cos(z) \cdot  \frac {\partial z} {\partial x}  + y\end{align*}

\textbf{Дифференцирование формулы:}
\begin{align*}
 \frac {\partial^2 f} {\partial x \cdot \partial y}  = (0 - \sin(z) \cdot  \frac {\partial z} {\partial y} ) \cdot  \frac {\partial z} {\partial x}  + \cos(z) \cdot  \frac {\partial^2 z} {\partial x \cdot \partial y}  + 1\end{align*}

\textbf{И снова оптимизация формулы:}
\begin{align*}
 \frac {\partial^2 f} {\partial x \cdot \partial y}  = (0 - \sin(z) \cdot  \frac {\partial z} {\partial y} ) \cdot  \frac {\partial z} {\partial x}  + \cos(z) \cdot  \frac {\partial^2 z} {\partial x \cdot \partial y}  + 1\end{align*}
